%matplotlib inline

dir='/media/secure_volume/hast2/'
file_warpoem='war_preprocessed.txt'

#read war poems
with open(dir+file_warpoem) as f:
    text=f.read()
    text2=re.split(r'\n---\d',text[1:])
war_poems=[]
text2[0]=text2[0][4:]
for i in text2:
    k=0
    j=['1'+i[k:k+3],i[k+3:]]
    war_poems.append(j)
war_poems_all=''
for i in war_poems:
    war_poems_all+="\n"+i[1]


#preprocessing
from nltk.stem.porter import PorterStemmer as PS
ps=PS()
def preprocess_1(text=war_poems_all):
    text=text.lower()
    text=re.sub('[.,;:!?*+^\[\]\{\}\"\#\$\%\&\'\(\)\-\—\_]',' ',text)
    return text

def preprocess_2(word=word):
    word=ps.stem(word.rstrip())
    return word


#word count
words_wp={}
text=preprocess_1(war_poems_all)
sum_wp=0
for word in text.split():
    word=preprocess_2(word)
#    word=ps.stem(word.rstrip())
    words_wp[word]=words_wp.get(word,0)+1
    sum_wp+=1


#percentage wp
words_wp2={}
for i in words_wp:
    words_wp2[i]=words_wp[i]/sum_wp


#read htrc poems
dir2='/media/secure_volume/workset'
words_htrc={}
sum_htrc=0
for v in tqdm.tqdm(glob.glob(dir2+'/**/*.txt', recursive=True)):
    with open(v) as f:
        #print(v)
        text=f.read()
        if 'volume-rights.txt' in v:
            text=''
            print('skipped')
        text=preprocess_1(text)
        #count htrc
        for word in text.split():
            word=preprocess_2(word)
#            word=ps.stem(word.rstrip())
            words_htrc[word]=words_htrc.get(word,0)+1
            sum_htrc+=1
#percentage htrc
words_htrc2={}
for i in tqdm.tqdm(words_htrc):
    words_htrc2[i]=words_htrc[i]/sum_htrc


#save dictionary
file_dic='vocab_dic.pkl'
